// Builds the Angel Artic logo set. Usage: RESVG=/path/to/@resvg/resvg-js node scripts/build-logo.mjs
// Writes SVG sources to assets/brand and PNGs to assets/ and store/brand.
import { writeFileSync, mkdirSync } from 'node:fs';
const { Resvg } = await import(process.env.RESVG ?? '@resvg/resvg-js');
const FONT = 'node_modules/@expo-google-fonts/baloo-2/800ExtraBold/Baloo2_800ExtraBold.ttf';
const FONT2 = 'node_modules/@expo-google-fonts/nunito/700Bold/Nunito_700Bold.ttf';

const defs = (id) => `
<defs>
  <linearGradient id="bg${id}" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="#FF9A8B"/><stop offset=".5" stop-color="#FF6B6B"/><stop offset="1" stop-color="#E4527F"/></linearGradient>
  <radialGradient id="glow${id}" cx=".5" cy=".46" r=".5"><stop offset="0" stop-color="#fff" stop-opacity=".45"/><stop offset="1" stop-color="#fff" stop-opacity="0"/></radialGradient>
  <linearGradient id="skin${id}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#FFE8D2"/><stop offset="1" stop-color="#FFD0AA"/></linearGradient>
  <linearGradient id="robe${id}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#fff"/><stop offset="1" stop-color="#E9E3FF"/></linearGradient>
  <linearGradient id="wing${id}" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="#fff"/><stop offset="1" stop-color="#E4DDFF"/></linearGradient>
  <linearGradient id="gold${id}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#FFE98A"/><stop offset="1" stop-color="#FFB81F"/></linearGradient>
  <linearGradient id="hair${id}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#8F80FF"/><stop offset="1" stop-color="#5B49D6"/></linearGradient>
  <filter id="sh${id}" x="-30%" y="-60%" width="160%" height="200%"><feDropShadow dx="0" dy="14" stdDeviation="14" flood-color="#7A1F3D" flood-opacity=".28"/></filter>
  <filter id="gl${id}" x="-40%" y="-150%" width="180%" height="400%"><feGaussianBlur stdDeviation="14"/></filter>
</defs>`;

const feather = 'M0 0 C70 -64 250 -78 338 -24 C350 -8 349 14 334 28 C256 76 86 74 0 0Z';
const star = (x, y, r, c = '#FFE98A') => `<path d="M${x} ${y - r} Q${x + r * 0.18} ${y - r * 0.18} ${x + r} ${y} Q${x + r * 0.18} ${y + r * 0.18} ${x} ${y + r} Q${x - r * 0.18} ${y + r * 0.18} ${x - r} ${y} Q${x - r * 0.18} ${y - r * 0.18} ${x} ${y - r}Z" fill="${c}"/>`;

/** The angel mark on a 1024 canvas. mono = flat white silhouette (Android themed icon). */
function mark(id, mono = false, standalone = false) {
  const f = (v, m = '#fff') => (mono ? m : v);
  const wingFill = f(`url(#wing${id})`);
  const wingStroke = mono ? 'none' : '#DCD3FA';
  const wing = `
    <g fill="${wingFill}" stroke="${wingStroke}" stroke-width="4" stroke-linejoin="round">
      <path d="${feather}" transform="translate(420 690) rotate(8) scale(-0.82 0.9)"/>
      <path d="${feather}" transform="translate(420 662) rotate(28) scale(-0.98 0.98)"/>
      <path d="${feather}" transform="translate(420 634) rotate(48) scale(-1.06 1.02)"/>
      <path d="${feather}" transform="translate(420 606) rotate(68) scale(-1.0 1.0)"/>
    </g>`;
  return `
  <g ${mono ? '' : `filter="url(#sh${id})"`}>
    ${wing}
    <g transform="translate(1024 0) scale(-1 1)">${wing}</g>
    <path d="${standalone ? 'M240 930 C240 810 364 712 512 712 C660 712 784 810 784 930 Q784 962 752 962 L272 962 Q240 962 240 930Z' : 'M232 1040 C232 810 360 712 512 712 C664 712 792 810 792 1040Z'}" fill="${f(`url(#robe${id})`)}"/>
    <circle cx="512" cy="492" r="208" fill="${f(`url(#skin${id})`)}"/>
  </g>
  ${mono ? '' : `
  <path d="M440 722 Q512 780 584 722" fill="none" stroke="#CFC5F5" stroke-width="10" stroke-linecap="round"/>
  <path d="M303 478 C292 308 408 252 512 252 C616 252 732 308 721 478 C664 406 592 368 512 368 C432 368 360 406 303 478Z" fill="url(#hair${id})"/>
  <path d="M372 372 C400 318 450 292 506 288" fill="none" stroke="#fff" stroke-opacity=".28" stroke-width="16" stroke-linecap="round"/>
  <ellipse cx="430" cy="500" rx="27" ry="32" fill="#2D2A4A"/><ellipse cx="594" cy="500" rx="27" ry="32" fill="#2D2A4A"/>
  <circle cx="440" cy="487" r="10" fill="#fff"/><circle cx="604" cy="487" r="10" fill="#fff"/>
  <circle cx="382" cy="566" r="35" fill="#FF9DB0" opacity=".7"/><circle cx="642" cy="566" r="35" fill="#FF9DB0" opacity=".7"/>
  <path d="M446 574 Q512 668 578 574 Q512 594 446 574Z" fill="#7A2E48" stroke="#2D2A4A" stroke-width="12" stroke-linejoin="round"/>
  <path d="M482 620 Q512 640 542 620 Q512 604 482 620Z" fill="#FF8FAB"/>`}
  <g ${mono ? '' : `filter="url(#sh${id})"`}>
    ${mono ? '' : `<ellipse cx="512" cy="206" rx="146" ry="40" fill="none" stroke="#FFD24A" stroke-width="34" opacity=".7" filter="url(#gl${id})"/>`}
    <ellipse cx="512" cy="206" rx="146" ry="40" fill="none" stroke="${f(`url(#gold${id})`)}" stroke-width="30"/>
  </g>
  ${mono ? '' : `${star(176, 250, 44)}${star(858, 330, 34, '#fff')}${star(826, 168, 22)}${star(238, 420, 20, '#fff')}`}`;
}

const svgIcon = (extra = '') => `<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024" viewBox="0 0 1024 1024">${defs('a')}
<rect width="1024" height="1024" fill="url(#bga)"/><rect width="1024" height="1024" fill="url(#gloa)"/>${mark('a')}${extra}</svg>`;
const wrapMark = (inner, scale, mono = false) => `<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024" viewBox="0 0 1024 1024">${defs('m')}<g transform="translate(${512 - 512 * scale} ${512 - 540 * scale}) scale(${scale})">${inner}</g></svg>`;

const render = (svg, w, file, opts = {}) => {
  const r = new Resvg(svg, { fitTo: { mode: 'width', value: w }, font: { fontFiles: [FONT, FONT2], loadSystemFonts: false, defaultFontFamily: 'Baloo 2' }, ...opts });
  writeFileSync(file, r.render().asPng());
};

mkdirSync('assets/brand', { recursive: true }); mkdirSync('store/brand', { recursive: true });
const icon = svgIcon().replace('<rect width="1024" height="1024" fill="url(#gloa)"/>', '<circle cx="512" cy="470" r="470" fill="url(#gloa)"/>');
writeFileSync('assets/brand/icon.svg', icon);
render(icon, 1024, 'assets/icon.png');
render(icon, 96, 'assets/favicon.png');

const markSvg = wrapMark(mark('m', false, true), 0.9);
writeFileSync('assets/brand/mark.svg', markSvg);
render(wrapMark(mark('m', false, true), 0.78), 1024, 'assets/splash-icon.png');
render(markSvg, 1024, 'store/brand/logo-mark.png');
render(wrapMark(mark('m', false, true), 0.62), 1024, 'assets/android-icon-foreground.png');
render(wrapMark(mark('m', true, true), 0.62, true), 1024, 'assets/android-icon-monochrome.png');
render(`<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024">${defs('b')}<rect width="1024" height="1024" fill="url(#bgb)"/><circle cx="512" cy="470" r="470" fill="url(#glob)"/></svg>`, 1024, 'assets/android-icon-background.png');

<rect width="2000" height="720" fill="#FFF9F0"/>
<clipPath id="clipH"><rect width="1024" height="1024" rx="230"/></clipPath><g transform="translate(40 40) scale(.64)"><g clip-path="url(#clipH)"><rect width="1024" height="1024" fill="url(#bgh)"/><circle cx="512" cy="470" r="470" fill="url(#gloh)"/>${mark('h')}</g></g>
<text x="760" y="360" font-family="Baloo 2" font-size="190" fill="#2D2A4A">Angel <tspan fill="#FF6B6B">Artic</tspan></text>

<text x="766" y="560" font-family="Nunito" font-weight="700" font-size="44" fill="#7A7799">Gentle speech sound practice for children</text></svg>`;
